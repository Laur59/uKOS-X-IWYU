# Host (native) unit tests for uKOS-X CLI modules.
#
# The modules under test are compiled UNMODIFIED, with a force-included shim
# that neutralises the three things a Mach-O host cannot take. Drive this
# through Tools/Developer/bin/run-tests, which also runs the drift checks.

include(${CMAKE_CURRENT_LIST_DIR}/cmake/HostCompiler.cmake)     # MUST precede project()

project(ukos_host_tests LANGUAGES C)

    message(FATAL_ERROR "UKOS_ROOT does not look like a uKOS-X tree: ${UKOS_ROOT}")
endif()

# The production warning set (Ports/cmake/system.cmake), minus -fshort-enums
# (which would diverge the enum ABI from the host libc) and -fstack-usage.

set(UKOS_HOST_WARNINGS
    -std=c23
    -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Wunreachable-code
    -Wswitch-default -Wcast-align -Wcast-qual -Wstrict-prototypes -Wc++-compat)

# uKOS.c includes version.h, which the system build generates into each
# variant's build directory from the git description. Generate a fixed one the
# same way rather than tracking a file - .gitignore excludes version.h, and
# correctly so. Fixed on purpose: a module that prints the firmware revision
# can then be asserted against a known string.

set(UKOS_TESTS_GENERATED ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${UKOS_TESTS_GENERATED})
file(WRITE ${UKOS_TESTS_GENERATED}/version.h
"/* Generated by Tools/Developer/tests/CMakeLists.txt. Do not edit by hand. */\n"
"#ifndef UKOS_HOST_VERSION_H\n"
"#define UKOS_HOST_VERSION_H\n"
"#define VERSIONING_TOOL     \"git:\"\n"
"#define SW_VERSION          \"host-test-0000000\"\n"
"#endif\n")

# ukos_fakes_text.c is a separate translation unit on purpose - see its header.

    harness/ukos_test.c
    harness/ukos_test_main.c

    target_compile_options(ukos_test_support PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer -g)
    target_link_options   (ukos_test_support PUBLIC -fsanitize=address,undefined)
endif()

    target_compile_options(ukos_port_support PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer -g)

# Tier 3 - the real kernel. Same stub port headers as tier 2, but it must NOT
# link ukos_port_support: that library DEFINES vKern_mutx, vKern_sema,
# vKern_proc, vKern_mbox, vKern_sign and fourteen counters, which are exactly
# what mutexes.c, semaphores.c, processes.c, mailboxes.c and signals.c define
# themselves. Keeping them in separate libraries makes the clash a duplicate
# symbol at link rather than a silent substitution, and it is why
# ukos_fake_runningCore is defined twice in this tree - once per tier.

add_library(ukos_kern_support STATIC fakes/ukos_fakes_kernenv.c)
target_include_directories(ukos_kern_support PUBLIC ${UKOS_PORT_STUB_INCLUDES})
target_link_libraries     (ukos_kern_support PUBLIC ukos_test_support)
target_compile_options    (ukos_kern_support PRIVATE ${UKOS_HOST_WARNINGS} -Werror)

if(UKOS_TESTS_SANITIZE)
    target_compile_options(ukos_kern_support PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer -g)
endif()

# ukos_add_cli_module_test(NAME <target> MODULE_SOURCE <path> TEST_SOURCE <path> [DEFINES ...])
#
# Adding another CLI module later is one more call plus its test file.

function(ukos_add_cli_module_test)
    cmake_parse_arguments(ARG "PORT_STUB" "NAME;MODULE_SOURCE;TEST_SOURCE" "DEFINES;MODULE_OPTIONS" ${ARGN})

    add_executable(${ARG_NAME} ${ARG_TEST_SOURCE} ${ARG_MODULE_SOURCE})

    if(ARG_PORT_STUB)
        target_link_libraries(${ARG_NAME} PRIVATE ukos_port_support)
    else()
        target_link_libraries(${ARG_NAME} PRIVATE ukos_test_support)
    endif()

    if(ARG_DEFINES)
        target_compile_definitions(${ARG_NAME} PRIVATE ${ARG_DEFINES})
    endif()

    # The module under test: force-include the shim, and scope away the one
    # host-only diagnostic (date.c:153, signed time_t x unsigned clock_t on
    # macOS). Deliberately NOT silenced globally - the same expression wraps for
    # pre-1970 dates, which the tests cover instead.

    set_source_files_properties(${ARG_MODULE_SOURCE} DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} PROPERTIES
        COMPILE_OPTIONS "-include;${CMAKE_CURRENT_SOURCE_DIR}/shim/ukos_host_shim.h;-Wno-sign-conversion;${ARG_MODULE_OPTIONS}"
        OBJECT_DEPENDS  "${CMAKE_CURRENT_SOURCE_DIR}/shim/ukos_host_shim.h")

    # Our own code is held to -Werror.

    set_source_files_properties(${ARG_TEST_SOURCE} DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} PROPERTIES
        COMPILE_OPTIONS "-Werror")

    add_test(NAME ${ARG_NAME} COMMAND ${ARG_NAME})

    # A hostile default timezone, so a test that forgets to pin TZ fails loudly
    # instead of silently inheriting the developer's zone.

    set_tests_properties(${ARG_NAME} PROPERTIES
        ENVIRONMENT "TZ=Pacific/Kiritimati;LC_ALL=C;UBSAN_OPTIONS=print_stacktrace=1")
endfunction()

# Tier 3 - kernel suites
# ======================
#
# ukos_add_kern_module_test(NAME <target> [KERN_SOURCES <path>...]
#                           TEST_SOURCE <path> [DEFINES ...] [MODULE_OPTIONS ...])
#
# Two differences from ukos_add_cli_module_test:
#
#   - several kernel sources per executable, because the kernel modules depend on
#     each other (pools is built on identifier, and later on semaphores) and the
#     real file is preferred to a fake wherever it will link.
#   - NO shim. The shim exists to rewrite MODULE() for Mach-O and to capture
#     dprintf; kernel files have neither, so forcing it in would only add a
#     divergence to explain.

function(ukos_add_kern_module_test)
    cmake_parse_arguments(ARG "" "NAME;TEST_SOURCE" "KERN_SOURCES;FAKE_SOURCES;DEFINES;MODULE_OPTIONS" ${ARGN})

    # FAKE_SOURCES are doubles that cannot live in ukos_kern_support because a
    # suite might compile the real thing instead - the four kern_*Semaphore
    # functions being the case in point: pools needs them faked, semaphores
    # needs them real, and one library cannot serve both.

    add_executable(${ARG_NAME} ${ARG_TEST_SOURCE} ${ARG_KERN_SOURCES} ${ARG_FAKE_SOURCES})
    target_link_libraries(${ARG_NAME} PRIVATE ukos_kern_support)

    # OS/Includes/debug.h #errors unless one architecture macro is defined, and
    # several kernel modules include it for DEBUG_KERN_TRACE. CORTEX_M7_S to
    # match port/macros_core.h, whose PRIVILEGE_* block is a drift-checked copy
    # of that core's. It is inert beyond the guard: CORTEX_M7_S appears nowhere
    # else in OS/, and the uKOS_ARCHITECTURE it selects is read only by a
    # coredump model this tier never compiles.

    target_compile_definitions(${ARG_NAME} PRIVATE CORTEX_M7_S)

    if(ARG_DEFINES)
        target_compile_definitions(${ARG_NAME} PRIVATE ${ARG_DEFINES})
    endif()

    # The kernel sources are compiled exactly as shipped. They get the production
    # warning set through the library's PUBLIC options but are not held to
    # -Werror, same as a CLI module under test.

    if(ARG_KERN_SOURCES AND ARG_MODULE_OPTIONS)
        set_source_files_properties(${ARG_KERN_SOURCES} DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} PROPERTIES
            COMPILE_OPTIONS "${ARG_MODULE_OPTIONS}")
    endif()

    set_source_files_properties(${ARG_TEST_SOURCE} DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} PROPERTIES
        COMPILE_OPTIONS "-Werror")

    add_test(NAME ${ARG_NAME} COMMAND ${ARG_NAME})

    set_tests_properties(${ARG_NAME} PROPERTIES
        ENVIRONMENT "TZ=Pacific/Kiritimati;LC_ALL=C;UBSAN_OPTIONS=print_stacktrace=1")
endfunction()
